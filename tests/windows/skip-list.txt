# Windows skip-list (--skipfile). Empty lines and # comments are ignored.
# A pattern is a full test name, or a regexp when it starts with "/".
# Only facilities Windows does not provide belong here. Replication, cluster,
# blocking pops, SCAN, KEYS, RANDOMKEY, SORT, MIGRATE, FLUSH/SWAPDB,
# query-buffer shrink, script and function kill, MONITOR, and extra clients run.

# Process stop/continue that the test itself requires as SIGSTOP/SIGCONT
# (pause/resume of a stopped client, DEBUG SLEEP while stopped).
# pause_process in tests/support/util.tcl is implemented via the test
# launcher on Windows and those tests are not named like the patterns below.
/pause process
/resume process
/DEBUG SLEEP.*stopped
/stopped clients

# Abstract-namespace Unix sockets (not implemented by Windows AF_UNIX).
/abstract unix

# Linux /proc and Linux-only process controls.
/smaps
/taskset
/setsid
/daemonize
